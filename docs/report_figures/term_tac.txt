=== Three-Address Code (TAC) ===
func main() -> int:
    alloc scale : int
    scale = 8
    alloc total : int
    total = 0
    alloc i : int
    i = 0
L_while_start0:
    t0 = i < 1000
    ifFalse t0 goto L_while_end1
    t1 = i * scale
    t2 = total + t1
    t3 = i * scale
    t4 = t2 + t3
    total = t4
    t5 = i + 1
    i = t5
    goto L_while_start0
L_while_end1:
    t6 = total % 256
    return t6
